\section*{Problem 4}
\addcontentsline{toc}{section}{Problem 4}
\markboth{Problem 4}{Problem 4}

\textbf{Problem Statement (Textbook 4.4.8):} 

Define
$$\|A\| = \sum_{i=1}^{n} \sum_{j=1}^{n} |a_{ij}|$$

Show that this is a matrix norm (that is, a norm on the linear space of all $n \times n$ matrices). Show that it is not subordinate to any vector norm. Does it conform to Equation (9) and Inequality (10)?

$$\|I\| = 1 \quad \text{(Equation 9)}$$
$$\|AB\| \leq \|A\|\|B\| \quad \text{(Inequality 10)}$$

\noindent\rule{\textwidth}{0.4pt}

\vspace{1em}

\textbf{Solution:}

\textbf{Part 1: Show $\|A\|$ is a matrix norm}

A matrix norm must satisfy three postulates:

\textbf{(1) Positivity:} If $A \neq 0$, then $\exists$ some $a_{ij} \neq 0$, so $|a_{ij}| > 0$. Thus $\|A\| = \sum_{i,j} |a_{ij}| > 0$.

\textbf{(2) Homogeneity:} $\|\lambda A\| = \sum_{i,j} |\lambda a_{ij}| = \sum_{i,j} |\lambda| |a_{ij}| = |\lambda| \sum_{i,j} |a_{ij}| = |\lambda| \|A\|$.

\textbf{(3) Triangle inequality:} 
\begin{align*}
\|A + B\| &= \sum_{i,j} |a_{ij} + b_{ij}| \leq \sum_{i,j} (|a_{ij}| + |b_{ij}|) \\
&= \sum_{i,j} |a_{ij}| + \sum_{i,j} |b_{ij}| = \|A\| + \|B\|
\end{align*}

$\boxed{\text{Therefore, } \|A\| \text{ is a matrix norm.}}$

\vspace{0.5em}

\textbf{Part 2: Show it is NOT subordinate to any vector norm}

A subordinate (induced) matrix norm satisfies $\|I\| = 1$. However, for the identity matrix $I$ where $(I)_{ij} = \delta_{ij}$ (Kronecker delta: $\delta_{ij} = 1$ if $i = j$, else $0$):
$$\|I\| = \sum_{i=1}^{n} \sum_{j=1}^{n} |\delta_{ij}| = \sum_{i=1}^{n} 1 = n$$

Since $\|I\| = n \neq 1$ for $n > 1$, $\boxed{\text{this norm is NOT subordinate to any vector norm.}}$

\vspace{0.5em}

\textbf{Part 3: Does it conform to Equation (9)?}

Equation (9) requires $\|I\| = 1$. As shown above, $\|I\| = n$.

$\boxed{\text{No, it does NOT conform to Equation (9).}}$

\vspace{0.5em}

\textbf{Part 4: Does it conform to Inequality (10)?}

We need to check if $\|AB\| \leq \|A\| \|B\|$.

\textit{Step 1:} We recall that for matrix multiplication, the $(i,k)$ entry of $AB$ is:
$$(AB)_{ik} = \sum_{j=1}^{n} a_{ij}b_{jk}$$

\textit{Step 2:} Using the triangle inequality for absolute values, we have:
$$|(AB)_{ik}| = \left|\sum_{j=1}^{n} a_{ij}b_{jk}\right| \leq \sum_{j=1}^{n} |a_{ij}b_{jk}| = \sum_{j=1}^{n} |a_{ij}| |b_{jk}|$$

\textit{Step 3:} We now compute $\|AB\|$ by summing over all entries:
\begin{align*}
\|AB\| &= \sum_{i=1}^{n} \sum_{k=1}^{n} |(AB)_{ik}| \\
&\leq \sum_{i=1}^{n} \sum_{k=1}^{n} \sum_{j=1}^{n} |a_{ij}| |b_{jk}| \quad \text{(from Step 2)}
\end{align*}

\textit{Step 4:} We rearrange the triple sum. Notice that $|a_{ij}|$ doesn't depend on $k$, and $|b_{jk}|$ doesn't depend on $i$:
\begin{align*}
\sum_{i=1}^{n} \sum_{k=1}^{n} \sum_{j=1}^{n} |a_{ij}| |b_{jk}| &= \sum_{i=1}^{n} \sum_{j=1}^{n} |a_{ij}| \left(\sum_{k=1}^{n} |b_{jk}|\right) \\
&= \sum_{i=1}^{n} \sum_{j=1}^{n} |a_{ij}| \cdot \sum_{k=1}^{n} \sum_{j=1}^{n} |b_{jk}| \\
&= \left(\sum_{i=1}^{n} \sum_{j=1}^{n} |a_{ij}|\right) \left(\sum_{j=1}^{n} \sum_{k=1}^{n} |b_{jk}|\right) \\
&= \|A\| \cdot \|B\|
\end{align*}

\textit{Step 5:} Combining Steps 3 and 4, we obtain: $\|AB\| \leq \|A\| \|B\|$.

$\boxed{\text{Yes, it DOES conform to Inequality (10).}}$
