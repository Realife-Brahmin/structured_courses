\section{Appendix: Derivation of Trapezoidal Discretization for Capacitor} \label{sec:appendix_cap_trapz}

To derive the equations \( v_{t} \) and \( i_{t} \), the following steps are involved:

\begin{itemize}
    \item Start with the governing equation for a capacitor:
    \begin{equation}
        i(t) = C \frac{dv(t)}{dt}
    \end{equation}
    \item Apply the trapezoidal rule to discretize the relationship between voltage and current:
    \begin{equation}
        v(t) - v(t - \Delta t) = \frac{\Delta t}{2C} \left[ i(t) + i(t - \Delta t) \right]
    \end{equation}
    \item Define the equivalent resistance \( R_C \) and history terms:
    \begin{equation}
        R_C = \frac{\Delta t}{2C}
    \end{equation}
    \item Derive the Thevenin (series) and Norton (parallel) companion models.
\end{itemize}

The final equations for the capacitor are:

\begin{align}
    \Aboxed{v(t) &= e_h + R_C i(t)} \label{eq:vt_cap_trapz_app} \\
    \Aboxed{i(t) &= \frac{v(t)}{R_C} - i_h} \label{eq:it_cap_trapz_app}
\end{align}

where:
\begin{itemize}
    \item \( e_h = v(t - \Delta t) + R_C i(t - \Delta t) \)
    \item \( i_h = \frac{v(t - \Delta t)}{R_C} - i(t - \Delta t) \)
\end{itemize}

\section{Appendix: Derivation of Backward-Euler Discretization for Capacitor} \label{sec:appendix_cap_be}

Starting from the capacitor constitutive relation
\begin{equation}
    i(t) = C\frac{dv(t)}{dt},
\end{equation}
the Backward-Euler (BE) discretization at time $t$ approximates the derivative by
\begin{equation}
    \frac{dv(t)}{dt} \approx \frac{v(t) - v(t-\Delta t)}{\Delta t}.
\end{equation}
Substituting and rearranging yields
\begin{equation}
    i(t) \approx C\frac{v(t)-v(t-\Delta t)}{\Delta t}.
\end{equation}
Writing the BE companion in a Thévenin/Norton form gives an equivalent resistance and history terms. The final boxed companion equations used in the main text (capacitor, BE) are
\begin{align}
    \Aboxed{v(t) &= e_h + R_C\, i(t)} \label{eq:vt_cap_be_app} \\
    \Aboxed{i(t) &= \frac{v(t)}{R_C} - i_h} \label{eq:it_cap_be_app}
\end{align}
with the BE companion parameters (for the capacitor):
\begin{align}
    R_C &= \frac{\Delta t}{C}, \\
    e_h &= v(t-\Delta t), \\
    i_h &= \frac{v(t-\Delta t)}{R_C}.
\end{align}

These expressions are used when implementing the Backward-Euler discretization in the numerical examples and the BE table in the main text.

\section{Appendix: Derivation of Forward-Euler Discretization for Capacitor} \label{sec:appendix_cap_fe}

Starting from the capacitor constitutive relation
\begin{equation}
    i(t) = C\frac{dv(t)}{dt},
\end{equation}
the Forward-Euler (FE) discretization at time $t$ approximates the derivative using the previous time step:
\begin{equation}
    \frac{dv(t)}{dt} \approx \frac{v(t) - v(t-h)}{h}.
\end{equation}
Substituting and rearranging yields
\begin{equation}
    i(t) \approx C\frac{v(t)-v(t-h)}{h}.
\end{equation}

To cast this in a companion model form similar to BE and trapezoidal methods, we introduce an artificial small resistance $R_\epsilon = h/C$ (as justified in the main text to avoid division by zero). The final boxed companion equations used in the main text (capacitor, FE) are
\begin{align}
    \Aboxed{v(t) &= e_h(t)} \label{eq:vt_cap_fe_app} \\
    \Aboxed{i(t) &= \frac{v(t) - e_h(t)}{R_\epsilon}} \label{eq:it_cap_fe_app}
\end{align}
with the FE companion parameters (for the capacitor):
\begin{align}
    R_c &= \frac{h}{C} \quad, \\
    e_h(t) &= v(t-h) + R_c\, i(t-h), \\
    i_h(t) &= \frac{e_h(t)}{R_c}.
\end{align}

These expressions are used when implementing the Forward-Euler discretization in the numerical examples and the FE table in the main text.

\section{Appendix: Derivation of Trapezoidal Discretization for Inductor} \label{sec:appendix_ind_trapz}

Starting from the inductor constitutive relation
\begin{equation}
    v(t) = L\frac{di(t)}{dt},
\end{equation}
we integrate from $t-h$ to $t$ to obtain:
\begin{equation}
    \int_{t-h}^{t} v(\tau) \, d\tau = L \left[ i(t) - i(t-h) \right].
\end{equation}

The trapezoidal rule approximates the integral as:
\begin{equation}
    \int_{t-h}^{t} v(\tau) \, d\tau \approx \frac{h}{2} \left[ v(t-h) + v(t) \right].
\end{equation}

Substituting and rearranging:
\begin{equation}
    i(t) - i(t-h) = \frac{h}{2L} \left[ v(t-h) + v(t) \right].
\end{equation}

Multiply both sides by $\frac{2L}{h}$ and rearrange:
\begin{equation}
    \frac{2L}{h} \left( i(t) - i(t-h) \right) = v(t-h) + v(t).
\end{equation}

Define the equivalent resistance $R_L = \frac{2L}{h}$:
\begin{equation}
    R_L \left( i(t) - i(t-h) \right) = v(t-h) + v(t).
\end{equation}

Solving for $v(t)$:
\begin{equation}
    v(t) = R_L i(t) - \underbrace{\left[ R_L i(t-h) + v(t-h) \right]}_{e_h}.
\end{equation}

The companion model parameters are:
\begin{align}
    R_L &= \frac{2L}{h} \\
    e_h &= R_L i(t-h) + v(t-h) \\
    i_h &= \frac{e_h}{R_L} = i(t-h) + \frac{v(t-h)}{R_L}
\end{align}

The final boxed equations are:
\begin{align}
    \Aboxed{v(t) &= R_L i(t) - e_h} \label{eq:vt_ind_trapz_app} \\
    \Aboxed{i(t) &= \frac{v(t)}{R_L} + i_h} \label{eq:it_ind_trapz_app}
\end{align}

\section{Appendix: Derivation of Backward Euler Discretization for Inductor} \label{sec:appendix_ind_be}

Starting from the inductor constitutive relation
\begin{equation}
    v(t) = L\frac{di(t)}{dt},
\end{equation}
the Backward Euler discretization approximates the derivative at time $t$ as:
\begin{equation}
    \frac{di(t)}{dt} \approx \frac{i(t) - i(t-h)}{h}.
\end{equation}

Substituting:
\begin{equation}
    v(t) \approx L \frac{i(t) - i(t-h)}{h}.
\end{equation}

Rearranging:
\begin{equation}
    v(t) = \frac{L}{h} i(t) - \frac{L}{h} i(t-h).
\end{equation}

Define the equivalent resistance $R_L = \frac{L}{h}$:
\begin{equation}
    v(t) = R_L i(t) - \underbrace{R_L i(t-h)}_{e_h}.
\end{equation}

The companion model parameters are:
\begin{align}
    R_L &= \frac{L}{h} \\
    e_h &= R_L i(t-h) \\
    i_h &= i(t-h) = \frac{e_h}{R_L}
\end{align}

The final boxed equations are:
\begin{align}
    \Aboxed{v(t) &= R_L i(t) - e_h} \label{eq:vt_ind_be_app} \\
    \Aboxed{i(t) &= \frac{v(t)}{R_L} + i_h} \label{eq:it_ind_be_app}
\end{align}
