% --- Derivation of the state-space model for the connected filter (Fig. 1(b)) ---
Referring to \Cref{fig:q10_hand_drawn_circuit_connected}, choose the states
\[
x \triangleq \begin{bmatrix} i_{L1} & i_{L2} & v_C \end{bmatrix}^{\!\top}, \qquad u \triangleq v_{dc}.
\]

\textbf{Node voltage at the junction between $L_1$ and $L_2$  -- Let's call the node $n1$ (not shown in diagram)}

By KCL through the shunt $r_{\text{leak}}$,
\begin{align}
    i_{r_\ell} &= \frac{v_{n1}}{r_{\text{leak}}} = i_{L1}-i_{L2} \\
    v_{n1} &= r_{\text{leak}}\,(i_{L1}-i_{L2}). \label{eq:KCL_leak}
\end{align}

\textbf{KVL from the source to node $n1$:}
\begin{align}
    u - r_{L1} i_{L1} - L_1 \frac{di_{L1}}{dt} - v_{n1} &= 0. \label{eq:KVL_dc_to_n1}
\end{align}

\textbf{KVL from node $n1$ to the capacitor node:}
\begin{align}
    v_{n1} - r_{L2} i_{L2} - L_2 \frac{di_{L2}}{dt} - v_C &= 0. \label{eq:KVL_n1_to_C}
\end{align}

\textbf{KCL at the capacitor node:}
\begin{align}
    i_{L2} &= C\,\frac{dv_C}{dt} + \frac{v_C}{r_C}. \label{eq:KCL_C}
\end{align}

\textbf{Solve for the state derivatives.}
Using \ref{eq:KVL_dc_to_n1} in \ref{eq:KVL_n1_to_C}:
\begin{align}
    L_1 \frac{di_{L1}}{dt} &= u - r_{L1}i_{L1} - r_{\text{leak}}(i_{L1} - i_{L2}) \nonumber \\
    &\Rightarrow \frac{di_{L1}}{dt} = \frac{1}{L_1} \left(u - (r_{L1} + r_{\text{leak}}) i_{L1} + r_{\text{leak}} i_{L2}\right). \label{eq:current_L1}
\end{align}

Using \ref{eq:current_L1} in \ref{eq:KCL_C}:
\begin{align}
    L_2 \frac{di_{L2}}{dt} &= r_{\text{leak}}(i_{L1} - i_{L2}) - r_{L2}i_{L2} - v_C \nonumber \\
    &\Rightarrow \frac{di_{L2}}{dt} = \frac{1}{L_2} \left(r_{\text{leak}} i_{L1} - (r_{\text{leak}} + r_{L2}) i_{L2} - v_C\right). \label{eq:current_L2}
\end{align}

Rewriting \ref{eq:KCL_C}:
\begin{align}
    \frac{dv_C}{dt} &= \frac{1}{C} i_{L2} - \frac{1}{r_C C} v_C. \label{eq:voltage_C}
\end{align}

\textbf{State–space form.}
Collecting \ref{eq:current_L1}–\ref{eq:voltage_C} in $\dot x = A x + B u$ gives
\[
\boxed{
A =
\begin{bmatrix}
-\dfrac{r_{L1}+r_{\text{leak}}}{L_1} & \dfrac{r_{\text{leak}}}{L_1} & 0\\[10pt]
\dfrac{r_{\text{leak}}}{L_2} & -\dfrac{r_{\text{leak}}+r_{L2}}{L_2} & -\dfrac{1}{L_2}\\[10pt]
0 & \dfrac{1}{C} & -\dfrac{1}{r_C C}
\end{bmatrix},
\qquad
B =
\begin{bmatrix}
\dfrac{1}{L_1}\\[6pt] 0\\[6pt] 0
\end{bmatrix}.
}
\]

\vspace{0.5em}
\textbf{Comment on numerical stiffness.}
The system is strongly stiff: the ratio between the fastest and the slow
time scales is
\[
\kappa \;\equiv\; \frac{|\Re\{\lambda_{\text{fast}}\}|}{|\Re\{\lambda_{\text{slow}}\}|}
\;\approx\; \frac{3.17\times 10^{7}}{28.2} \;\approx\; 1.1\times 10^{6}.
\]
One real pole at $-3.17\times 10^{7}$~s$^{-1}$ (time constant $\tau_{\text{fast}}\approx
31.6$~ns) coexists with a lightly–damped oscillatory pair
($\omega_d\approx 2533$~rad/s). This wide separation
forces very small steps for explicit fixed–step solvers.

% --- Simulation note & figure reference --------------------------------
\textbf{Fixed–step simulation.}
Using a conservative explicit step such as $h=10^{-4}$~s (0.1~ms) with a
5th–order explicit method produced stable, convergent waveforms for
$i_{L1}$, $i_{L2}$ and $v_C$ over $0.2$~s (see
Fig.~\ref{fig:q10_fixedstep_plots}). Larger $h$ begins to corrupt the phase
and envelope due to the lightly–damped oscillatory pair.

Using the implicit solver \verb|ode14x|, a maximum step size of $h=10^{-4}$~s (0.1~ms) produced stable, convergent waveforms for $i_{L1}$, $i_{L2}$, and $v_C$ over $0.2$~s (see Fig.~\ref{fig:q10_fixedstep_plots}). However, the quality of the solution begins to degrade at this upper limit of $h$. A smaller step size of $h=10^{-5}$~s (not shown here) yielded significantly better results, with improved accuracy and stability in the waveforms.

\subsection*{Stability Analysis Observations}
The stability regions for Forward Euler (ode1) and explicit RK5 (ode5) methods are shown in \Cref{fig:q10_stability_regions}. The scaled eigenvalues $z = h \lambda$ are plotted for various step sizes $h$, with green crosses indicating stable values and red crosses indicating unstable values.

The analysis reveals the following key insights:
\begin{itemize}
    \item The high negative eigenvalue ($\lambda \approx -3.17 \times 10^7$) primarily drives the need for a very small step size $h$. This is evident from the outer crosses in the plot, which correspond to this eigenvalue and are located far outside the stability region for larger $h$ values.
    \item The two smaller eigenvalues ($\lambda \approx -28.2 \pm j\,2533.3$) are encompassed within a very small region close to $z = 0$. These eigenvalues allow for slightly larger $h$ values, but the stability region remains limited due to the influence of the larger eigenvalue.
    \item For Forward Euler (ode1), the stability region is extremely small, requiring $h \lesssim 0.038~\mu\text{s}$ for stability. Explicit RK5 (ode5) offers a larger stability region, allowing $h \lesssim 0.162~\mu\text{s}$, but still requires careful tuning due to the stiffness of the system.
    \item The stiffness of the system, driven by the wide separation between the eigenvalues, makes implicit solvers like \verb|ode14x| more suitable for larger step sizes.
\end{itemize}

\subsection*{Maximum Stable Step Size}
From the plot and MATLAB output:
\begin{itemize}
    \item \textbf{Euler:} $h_\text{max} \approx 0.063~\mu\text{s}$
    \item \textbf{Taylor-5:} $h_\text{max} \approx 0.101~\mu\text{s}$
\end{itemize}
These values indicate the estimated largest time step for which each method expects to remain numerically stable for the given system.

\subsection*{Summary}
The explicit RK5 (Taylor-5) method allows for a larger stable time step than Forward Euler, as seen from the larger stability region. However, the presence of the high negative eigenvalue significantly limits the step size for both methods, highlighting the stiffness of the system and the need for implicit solvers for efficient integration.

\subsection*{Conclusion}
Hand and MATLAB eigenvalues match exactly. The large stiffness index
($\kappa\!\sim\!10^{6}$) makes explicit fixed–step integration delicate;
practical accuracy for the oscillatory mode requires $\sim$20–30 samples per
cycle ($h\approx 0.08$–$0.12$~ms), whereas the fast pole would force a
nanosecond step if treated explicitly. Hence, for efficiency and robustness,
an implicit fixed–step solver is recommended if larger $h$ is desired.
