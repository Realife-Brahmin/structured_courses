\section*{Question 2}
\addcontentsline{toc}{section}{Question 2}

Consider the series $RLC$ circuit with parameters $r_L$, $L$, and $C$ as in Q1. The equations are:

	\textbf{KVL:}
\begin{align}
    V_{\rm dc} - r_L i_L - L \frac{di_L}{dt} - v_C = 0 \tag{1}
\end{align}
where $i_L$ is the inductor current and $v_C$ is the capacitor voltage.

	\textbf{Capacitor current:}
\begin{align}
    i_L = C \frac{dv_C}{dt} \tag{2}
\end{align}

	\textbf{Substitute (2) into (1):}
\begin{align*}
    V_{\rm dc} - r_L C \frac{dv_C}{dt} - L C \frac{d^2 v_C}{dt^2} - v_C = 0
\end{align*}

	\textbf{Rearrange to standard form:}
\begin{align*}
    L C \frac{d^2 v_C}{dt^2} + r_L C \frac{dv_C}{dt} + v_C = V_{\rm dc}
\end{align*}

Divide both sides by $L C$:
\begin{align*}
    \frac{d^2 v_C}{dt^2} + \frac{r_L}{L} \frac{dv_C}{dt} + \frac{1}{L C} v_C = \frac{V_{\rm dc}}{L C}
\end{align*}

This is a standard second-order nonhomogeneous ODE. The homogeneous solution is:
\begin{align*}
    \frac{d^2 v_C}{dt^2} + \frac{r_L}{L} \frac{dv_C}{dt} + \frac{1}{L C} v_C = 0
\end{align*}

	\textbf{Eigenvalue computation:}
The characteristic equation is:
\begin{align*}
    s^2 + \frac{r_L}{L} s + \frac{1}{L C} = 0
\end{align*}
The eigenvalues are:
\begin{align*}
    s = \frac{-\frac{r_L}{L} \pm \sqrt{\left(\frac{r_L}{L}\right)^2 - 4 \cdot \frac{1}{L C}}}{2}
\end{align*}

For the given values ($r_L = 1~\Omega$, $L = 19~\text{mH}$, $C = 8.2~\mu\text{F}$):
\begin{align*}
    \frac{r_L}{L} &= \frac{1}{0.019} = 52.63~\text{s}^{-1} \\
    \frac{1}{L C} &= \frac{1}{0.019 \times 8.2 \times 10^{-6}} = 6,430,041~\text{s}^{-2}
\end{align*}

Plug into the quadratic formula:
\begin{align*}
    s = \frac{-52.63 \pm \sqrt{(52.63)^2 - 4 \times 6,430,041}}{2}
\end{align*}

The discriminant is negative, so the roots are complex conjugates:
\begin{align*}
    s = -26.316 \pm j\,2533.3
\end{align*}


    \begin{center}
    \fbox{\parbox{0.95\linewidth}{
        \textbf{Interpretation:} The eigenvalues are complex conjugates with negative real parts, confirming the circuit is underdamped. Based on the very low value of the damping ratio $\zeta = 0.0104$ calculated in Q1, the transient response will exhibit oscillatory behavior with a slow decay.
    }}
    \end{center}

\vspace{1em}
	\textbf{General solution:}
\begin{align*}
    v_C(t) = v_{C,\infty} + \left[v_{C,0} - v_{C,\infty}\right] \left[1 - \frac{1}{\sqrt{1-\zeta^2}} e^{-\zeta \omega_n t} \sin(\omega_d t + \phi)\right]
\end{align*}
where $v_{C,\infty}$ is the steady-state value, $v_{C,0}$ is the initial value, and $\omega_d = \omega_n \sqrt{1-\zeta^2}$.

For a step input, $v_{C,\infty} = x_{SS}$, and the solution simplifies to:
\[
\boxed{
    x(t) = x_{SS} \left[1 - \frac{1}{\sqrt{1-\zeta^2}} e^{-\zeta \omega_n t} \sin(\omega_d t + \phi)\right]
}
\]

For the given values ($r_L = 1~\Omega$, $L = 19~\text{mH}$, $C = 8.2~\mu\text{F}$), the numeric solution is:
\[
\boxed{
    x(t) [kV] = 11.3 * \left[1 - 1.00005 \, e^{-26.34\, t} \, \sin(2533.3\, t + 1.5604)\right]
}
\]
