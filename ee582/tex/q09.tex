% Question statement
In Fig. 1(a) the equivalent series resistance and inductance of the capacitor ($r_{L2}$ and $L_2$) were added to the inductance parameters $r_{L1}$ and $L_1$ (i.e. $r_L = r_{L1} + r_{L2}$ and $L = L_1 + L_2$). In Fig. 1(b), they are separated and also the capacitor leakage resistance ($r_c$) is considered. In addition, the discharge resistance ($r_{leak}$) is shown in Fig. 1(b). This resistance is used to discharge the filter capacitor when it is off-line for safety purposes which was ignored in the simple model shown in Fig. 1(a). Implement the model of Fig. 1(b) in Simulink using basic blocks considering $r_{L2} = 323~\text{m}\Omega$, $L_2 = 1~\text{mH}$, $r_c = 1~\text{M}\Omega$, and $r_{leak} = 30~\text{k}\Omega$. If the filter is disconnected, how long does it take for the capacitor to fully discharge?
\vspace{1em}

The derivation below refers to \Cref{fig:q09_hand_drawn_circuit} for analyzing the disconnected filter.
The capacitor $C$ discharges through its leakage resistance $r_C$ and the discharge resistor $r_{\text{leak}}$. The relevant equations can be derived as follows.

\vspace{1em}
\textbf{KCL at the capacitor node:}
\begin{align}
    i_{L2} = C \frac{dv_C}{dt} + \frac{v_C}{r_C} \tag{1}
\end{align}

\textbf{KVL across $L_2$ and resistances:}
\begin{align}
    -(r_{\text{leak}} + r_{L2}) i_{L2} - L_2 \frac{di_{L2}}{dt} = v_C \tag{2}
\end{align}

\textbf{Substitute (1) into (2):}
\begin{align*}
    -(r_{\text{leak}} + r_{L2}) \left(C \frac{dv_C}{dt} + \frac{v_C}{r_C}\right)
    - L_2 \left(C \frac{d^2 v_C}{dt^2} + \frac{1}{r_C}\frac{dv_C}{dt}\right) = v_C
\end{align*}

\textbf{Rearrange terms:}
\begin{align*}
    -L_2 C \frac{d^2 v_C}{dt^2}
    - \left[(r_{\text{leak}} + r_{L2}) C + \frac{L_2}{r_C}\right] \frac{dv_C}{dt}
    - \left(\frac{r_{\text{leak}} + r_{L2}}{r_C} + 1\right) v_C = 0
\end{align*}

\textbf{Multiply through by $-1$:}
\begin{align*}
    L_2 C \frac{d^2 v_C}{dt^2}
    + \left[(r_{\text{leak}} + r_{L2}) C + \frac{L_2}{r_C}\right] \frac{dv_C}{dt}
    + \left(\frac{r_{\text{leak}} + r_{L2}}{r_C} + 1\right) v_C = 0
\end{align*}

\textbf{Divide through by $L_2 C$:}
\begin{align*}
    \frac{d^2 v_C}{dt^2}
    + \left(\frac{r_{\text{leak}} + r_{L2}}{L_2} + \frac{1}{r_C C}\right) \frac{dv_C}{dt}
    + \frac{1}{L_2 C} \left(\frac{r_{\text{leak}} + r_{L2}}{r_C} + 1\right) v_C = 0
\end{align*}

\vspace{1em}
\textbf{Final second-order ODE (disconnected case):}
\[
\boxed{
    \frac{d^2 v_C}{dt^2}
    + \left(\frac{r_{\text{leak}} + r_{L2}}{L_2} + \frac{1}{r_C C}\right) \frac{dv_C}{dt}
    + \frac{1}{L_2 C} \left(\frac{r_{\text{leak}} + r_{L2}}{r_C} + 1\right) v_C = 0
}
\]

\vspace{1em}
This is a homogeneous second-order differential equation governing the capacitor voltage $v_C(t)$ during discharge when the filter is disconnected. The eigenvalues can be obtained by solving the characteristic quadratic:
\[
s^2 + \alpha s + \beta = 0,
\]
where
\[
\alpha = \frac{r_{\text{leak}} + r_{L2}}{L_2} + \frac{1}{r_C C}, 
\quad
\beta = \frac{1}{L_2 C} \left(\frac{r_{\text{leak}} + r_{L2}}{r_C} + 1\right).
\]
